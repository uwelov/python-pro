1"""Бізнес-логіка замовлень: створення, скасування, позначення оплаченим. 

2 

3Усі функції, що змінюють кілька таблиць (Order, OrderItem, Book.stock), працюють 

4в `transaction.atomic`: або виконується все, або нічого. Рядки Book блокуються 

5через `select_for_update`, щоб два одночасні замовлення не «продали» один і той 

6самий останній примірник. 

7"""

51def _lock_books(book_ids: Iterable[int]) -> dict[int, Book]: 

52 """Блокує рядки книг до кінця транзакції. Порядок за pk — щоб уникнути deadlock.""" 

53 books = Book.objects.select_for_update().filter(pk__in=list(book_ids)).order_by("pk") 

54 return {book.pk: book for book in books}

57@transaction.atomic 

58def create_order( 

59 *, 

60 user, 

61 full_name: str, 

62 email: str, 

63 phone: str = "", 

64 lines: Iterable[tuple[int, int]], 

65 notify: bool = True, 

66) -> Order: 

67 """Створює замовлення, фіксує ціни та резервує книги на складі. 

68 

69 Args: 

70 user: автор замовлення. 

71 full_name, email, phone: контактні дані. 

72 lines: послідовність (id_книги, кількість). 

73 notify: чи ставити лист клієнту в on_commit. Потік оформлення з оплатою 

74 передає False і шле лист сам, коли платіжну сесію вже створено. 

75 

76 Raises: 

77 EmptyOrderError: немає позицій. 

78 InsufficientStockError: якоїсь книги не вистачає. Транзакція відкочується 

79 повністю — не лишається ні замовлення, ні «зайвих» списань зі складу. 

80 

81 Лист клієнту ставиться в `transaction.on_commit`, тобто надсилається лише 

82 після успішного коміту (якщо транзакцію відкотило — листа не буде). 

83 """ 

84 lines = list(lines) 

85 if not lines: 

86 raise EmptyOrderError(_("Кошик порожній.")) 

87 

88 books = _lock_books(book_id for book_id, _ in lines) 

89 

90 order = Order.objects.create( 

91 user=user, full_name=full_name, email=email, phone=phone 

92 ) 

93 

94 items: list[OrderItem] = [] 

95 for book_id, quantity in lines: 

96 book = books.get(book_id) 

97 if book is None: 

98 raise InsufficientStockError(_("(книгу видалено)"), quantity, 0) 

99 if book.stock < quantity: 

100 raise InsufficientStockError(book.title, quantity, book.stock) 

101 

102 book.stock -= quantity 

103 book.save(update_fields=["stock"]) 

104 items.append( 

105 OrderItem( 

106 order=order, 

107 book=book, 

108 book_title=book.title, 

109 price=book.price, # snapshot ціни 

110 quantity=quantity, 

111 ) 

112 ) 

113 OrderItem.objects.bulk_create(items) 

114 

115 logger.info("Створено замовлення #%s (%s позицій)", order.pk, len(items)) 

116 if notify: 

117 transaction.on_commit(partial(send_order_created_email, order.pk)) 

118 return order

121@transaction.atomic 

122def cancel_order(order_id: int) -> bool: 

123 """Скасовує неоплачене замовлення й повертає книги на склад. 

124 

125 Returns: 

126 True, якщо скасовано; False, якщо замовлення вже не в стані "очікує оплати" 

127 (повторний виклик нічого не псує — функція ідемпотентна). 

128 """ 

129 order = Order.objects.select_for_update().get(pk=order_id) 

130 if order.status != Order.Status.PENDING: 

131 return False 

132 

133 items = list(order.items.exclude(book__isnull=True)) 

134 books = _lock_books(item.book_id for item in items) 

135 for item in items: 

136 book = books[item.book_id] 

137 book.stock += item.quantity 

138 book.save(update_fields=["stock"]) 

139 

140 order.status = Order.Status.CANCELLED 

141 order.save(update_fields=["status", "updated_at"]) 

142 logger.info("Замовлення #%s скасовано, склад відновлено", order.pk) 

143 return True

146@transaction.atomic 

147def cancel_stale_orders(*, older_than: timedelta) -> int: 

148 """Скасовує неоплачені замовлення, старші за `older_than`, і повертає книги на склад. 

149 

150 Страхує від ситуації, коли вебхук `checkout.session.expired` не дійшов (або не налаштований): 

151 без цього резерв на складі висів би вічно. Кожне замовлення скасовується окремо 

152 (`cancel_order` ідемпотентна), тож збій одного не блокує решту. 

153 

154 Returns: 

155 Кількість скасованих замовлень. 

156 """ 

157 cutoff = timezone.now() - older_than 

158 stale_ids = list( 

159 Order.objects.filter(status=Order.Status.PENDING, created_at__lt=cutoff).values_list("pk", flat=True) 

160 ) 

161 return sum(1 for order_id in stale_ids if cancel_order(order_id))

164@transaction.atomic 

165def mark_order_paid( 

166 order_id: int, 

167 *, 

168 session_id: str = "", 

169 payment_intent_id: str = "", 

170 amount_total: int | None = None, 

171 currency: str | None = None, 

172) -> bool: 

173 """Позначає замовлення оплаченим. Ідемпотентна: можна викликати з вебхука і зі сторінки успіху. 

174 

175 Args: 

176 order_id: id замовлення. 

177 session_id, payment_intent_id: ідентифікатори Stripe для запису в замовлення. 

178 amount_total: сума, яку фактично сплатили (в мінорних одиницях, з Stripe-сесії). 

179 currency: валюта платежу. Якщо `amount_total`/`currency` передано — вони мають збігатися 

180 із сумою та валютою замовлення (defense in depth: не довіряємо подіям «наосліп»). 

181 

182 Returns: 

183 True — статус змінено щойно; False — вже було оплачено, скасовано або сума/валюта не збіглись. 

184 """ 

185 order = Order.objects.select_for_update().get(pk=order_id) 

186 

187 if order.status == Order.Status.PAID: 

188 return False 

189 if order.status == Order.Status.CANCELLED: 

190 # Гроші прийшли за замовлення, яке вже скасоване (склад відпущено). 

191 # Автоматично не «воскрешаємо» — потрібне ручне рішення (повернення коштів). 

192 logger.error( 

193 "Оплата отримана за скасоване замовлення #%s — потрібне ручне втручання/refund", 

194 order.pk, 

195 ) 

196 return False 

197 

198 expected_currency = settings.STRIPE_CURRENCY 

199 expected_amount = payments.to_minor_units(order.get_total_cost(), expected_currency) 

200 if (amount_total is not None and amount_total != expected_amount) or ( 

201 currency is not None and currency.lower() != expected_currency 

202 ): 

203 logger.error( 

204 "Оплата за замовлення #%s не збігається із замовленням: отримано %s %s, очікувалось %s %s", 

205 order.pk, amount_total, currency, expected_amount, expected_currency, 

206 ) 

207 return False 

208 

209 order.status = Order.Status.PAID 

210 order.paid_at = timezone.now() 

211 if session_id: 

212 order.stripe_session_id = session_id 

213 if payment_intent_id: 

214 order.stripe_payment_intent_id = payment_intent_id 

215 order.save() 

216 logger.info("Замовлення #%s оплачено", order.pk) 

217 return True
