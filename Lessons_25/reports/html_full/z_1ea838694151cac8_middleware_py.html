18class RequestLoggingMiddleware: 

19 """Логує кожен HTTP-запит: метод, шлях, користувача, статус і тривалість. 

20 

21 Middleware «двоїстий» (sync + async): під WSGI працює синхронно, під ASGI — як 

22 корутина, тож не змушує Django загортати весь ланцюжок у потоки. В async-режимі 

23 користувача беремо через `await request.auser()`, бо `request.user.is_authenticated` 

24 може піти в БД синхронно, що в event loop заборонено. 

25 """ 

26 

27 sync_capable = True 

28 async_capable = True

30 def __init__(self, get_response) -> None: 

31 self.get_response = get_response 

32 if iscoroutinefunction(get_response): 

33 markcoroutinefunction(self)

35 def __call__(self, request: HttpRequest): 

36 if iscoroutinefunction(self): 

37 return self.__acall__(request) 

38 start = time.monotonic() 

39 response = self.get_response(request) 

40 user = getattr(request, "user", None) 

41 username = user.get_username() if user is not None and user.is_authenticated else "anonymous" 

42 self._log(request, response, start, username) 

43 return response
