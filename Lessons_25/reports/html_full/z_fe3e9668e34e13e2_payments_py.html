1"""Інтеграція зі Stripe Checkout (hosted-сторінка оплати). 

2 

3Потік оплати: 

4 1. створюємо Order (status=pending) і резервуємо книги; 

5 2. `create_checkout_session()` -> Stripe повертає session.url; 

6 3. редірект клієнта на session.url (оплата на боці Stripe); 

7 4. Stripe повертає клієнта на success_url і одночасно шле вебхук 

8 `checkout.session.completed` — за будь-яким із двох каналів 

9 замовлення позначається оплаченим (див. services.mark_order_paid). 

10"""

23# Валюти без копійок: сума передається цілим числом як є (решта — у мінорних одиницях, ×100). 

24ZERO_DECIMAL_CURRENCIES = frozenset( 

25 {"bif", "clp", "djf", "gnf", "jpy", "kmf", "krw", "mga", "pyg", "rwf",

30def to_minor_units(amount: Decimal, currency: str) -> int: 

31 """450.00 (uah) -> 45000; 450 (jpy) -> 450.""" 

32 if currency.lower() in ZERO_DECIMAL_CURRENCIES: 

33 return int(amount.quantize(Decimal("1"), rounding=ROUND_HALF_UP)) 

34 return int((amount * 100).quantize(Decimal("1"), rounding=ROUND_HALF_UP))

45def create_checkout_session(order: Order, *, success_url: str, cancel_url: str): 

46 """Створює Stripe Checkout Session для замовлення. 

47 

48 `success_url` має містити `{CHECKOUT_SESSION_ID}` — Stripe підставить id сесії. 

49 """ 

50 _configure() 

51 currency = settings.STRIPE_CURRENCY 

52 

53 line_items = [ 

54 { 

55 "quantity": item.quantity, 

56 "price_data": { 

57 "currency": currency, 

58 "unit_amount": to_minor_units(item.price, currency), 

59 "product_data": {"name": item.book_title}, 

60 }, 

61 } 

62 for item in order.items.all() 

63 ] 

64 

65 return stripe.checkout.Session.create( 

66 mode="payment", 

67 line_items=line_items, 

68 success_url=success_url, 

69 cancel_url=cancel_url, 

70 client_reference_id=str(order.pk), 

71 customer_email=order.email, 

72 metadata={"order_id": str(order.pk)}, 

73 payment_intent_data={"metadata": {"order_id": str(order.pk)}}, 

74 expires_at=int(time.time()) + settings.STRIPE_SESSION_TTL, 

75 )
