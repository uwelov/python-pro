1"""Асинхронні JSON API-views каталогу (async def + async ORM). 

2 

3Що тут показано: 

4 * `async for` по QuerySet і `acount()` / `aget()` / `aaggregate()` — асинхронний ORM; 

5 * `asyncio.gather()` — кілька запитів до БД «одночасно» в одному view; 

6 * `await request.auser()` / `await user.ahas_perm()` — асинхронна автентифікація. 

7 

8Важливо: у async-коді не можна звертатися до лінивих зв'язків (book.category) чи 

9`request.user` напряму — це синхронний доступ до БД і Django кине 

10SynchronousOnlyOperation. Тому `select_related()` та `await request.auser()`. 

11 

12Під WSGI (gunicorn) такі view теж працюють, але виконуються в окремому потоці; повноцінно 

13async-режим вмикається під ASGI-сервером (uvicorn) — саме так запускається Docker-образ. 

14"""

31def json_response(data, status: int = 200) -> JsonResponse: 

32 """JsonResponse, що не екранує кирилицю (\\u0414 -> Д) і приймає не лише dict.""" 

33 return JsonResponse( 

34 data, status=status, safe=False, json_dumps_params={"ensure_ascii": False}

38def serialize_book(book: Book) -> dict: 

39 """Книга -> dict. Потребує book.category, завантаженого через select_related.""" 

40 return { 

41 "id": book.pk, 

42 "title": book.title, 

43 "author": book.author, 

44 "price": str(book.price), 

45 "stock": book.stock, 

46 "in_stock": book.in_stock, 

47 "category": {"id": book.category_id, "name": book.category.name, "slug": book.category.slug}, 

48 }

51@require_GET 

52async def book_list_api(request: HttpRequest): 

53 """GET /api/books/?q=&category=&limit= — пошук книг (асинхронний ORM).""" 

54 queryset = Book.objects.select_related("category") 

55 

56 query = request.GET.get("q", "").strip() 

57 if query: 

58 queryset = queryset.filter(Q(title__icontains=query) | Q(author__icontains=query)) 

59 category = request.GET.get("category", "").strip() 

60 if category: 

61 queryset = queryset.filter(category__slug=category) 

62 

63 try: 

64 limit = min(max(int(request.GET.get("limit", 10)), 1), MAX_LIMIT) 

65 except ValueError: 

66 limit = 10 

67 

68 # Два незалежні запити виконуємо «одночасно». 

69 total, results = await asyncio.gather( 

70 queryset.acount(), 

71 _collect(queryset[:limit]), 

72 ) 

73 return json_response({"count": total, "results": results})

76async def _collect(queryset) -> list[dict]: 

77 """Асинхронно збирає QuerySet книг у список словників (`async for`). 

78 

79 Args: 

80 queryset: QuerySet книг із `select_related("category")` (інакше доступ до категорії 

81 в async-коді викличе SynchronousOnlyOperation). 

82 

83 Returns: 

84 Список результатів `serialize_book`.""" 

85 return [serialize_book(book) async for book in queryset]

88@require_GET 

89async def book_detail_api(request: HttpRequest, pk: int): 

90 """GET /api/books/<pk>/ — одна книга (`aget`).""" 

91 try: 

92 book = await Book.objects.select_related("category").aget(pk=pk) 

93 except Book.DoesNotExist: 

94 return json_response({"error": _("Книгу не знайдено.")}, status=404) 

95 return json_response(serialize_book(book))

98@require_GET 

99async def category_list_api(request: HttpRequest): 

100 """GET /api/categories/ — категорії з кількістю книг (annotate + async for).""" 

101 queryset = Category.objects.annotate(book_count=Count("books")).order_by("name") 

102 data = [ 

103 {"id": c.pk, "name": c.name, "slug": c.slug, "book_count": c.book_count} 

104 async for c in queryset 

105 ] 

106 return json_response(data)

109@require_GET 

110async def stats_api(request: HttpRequest): 

111 """GET /api/stats/ — зведена статистика: кілька запитів паралельно через gather.""" 

112 books_total, in_stock, categories, prices = await asyncio.gather( 

113 Book.objects.acount(), 

114 Book.objects.filter(stock__gt=0).acount(), 

115 Category.objects.acount(), 

116 Book.objects.aaggregate( 

117 avg=Avg("price"), min=Min("price"), max=Max("price"), units=Sum("stock") 

118 ), 

119 ) 

120 

121 def money(value): 

122 return None if value is None else str(round(value, 2)) 

123 

124 return json_response( 

125 { 

126 "books": books_total, 

127 "books_in_stock": in_stock, 

128 "books_out_of_stock": books_total - in_stock, 

129 "categories": categories, 

130 "units_in_stock": prices["units"] or 0, 

131 "price": {"avg": money(prices["avg"]), "min": money(prices["min"]), "max": money(prices["max"])}, 

132 } 

133 )

136@require_GET 

137async def low_stock_api(request: HttpRequest): 

138 """GET /api/low-stock/ — книги, що закінчуються. Лише для користувачів з правом change_book.""" 

139 user = await request.auser() # асинхронний аналог request.user 

140 if not user.is_authenticated: 

141 return json_response({"error": _("Потрібна автентифікація.")}, status=401) 

142 if not await user.ahas_perm("catalog.change_book"): 

143 return json_response({"error": _("Недостатньо прав.")}, status=403) 

144 

145 queryset = Book.objects.select_related("category").filter(stock__lte=LOW_STOCK_THRESHOLD).order_by("stock", "title") 

146 return json_response( 

147 {"threshold": LOW_STOCK_THRESHOLD, "results": [serialize_book(b) async for b in queryset]} 

148 )
