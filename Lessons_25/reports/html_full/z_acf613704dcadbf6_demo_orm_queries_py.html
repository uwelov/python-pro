1""" 

2Демонстрація Django ORM-запитів: filter, annotate, Q-об'єкти. 

3 

4Запуск (спочатку створити тестові дані): 

5 python manage.py seed_demo_data 

6 python manage.py demo_orm_queries 

7""" 

8 

9from __future__ import annotations 

10 

11from django.core.management.base import BaseCommand 

12from django.db.models import Avg, Count, Q 

13 

14from catalog.models import Book, Category 

15 

16 

17class Command(BaseCommand): 

18 help = "Виводить приклади ORM-запитів (filter, annotate, Q) у консоль." 

19 

20 def handle(self, *args, **options): 

21 self.stdout.write(self.style.MIGRATE_HEADING("\n1. filter() — книги в наявності дешевші за 500")) 

22 cheap_in_stock = Book.objects.filter(stock__gt=0, price__lt=500) 

23 for book in cheap_in_stock: 

24 self.stdout.write(f" - {book.title} ({book.price} грн, залишок {book.stock})") 

25 

26 self.stdout.write(self.style.MIGRATE_HEADING("\n2. filter() — книги певної категорії за slug")) 

27 it_books = Book.objects.filter(category__slug="it") 

28 for book in it_books: 

29 self.stdout.write(f" - {book.title}") 

30 

31 self.stdout.write( 

32 self.style.MIGRATE_HEADING( 

33 "\n3. Q-об'єкти — пошук за назвою АБО автором (регістронезалежно)" 

34 ) 

35 ) 

36 query = "мартін" 

37 search_results = Book.objects.filter( 

38 Q(title__icontains=query) | Q(author__icontains=query) 

39 ) 

40 for book in search_results: 

41 self.stdout.write(f" - {book.title} — {book.author}") 

42 

43 self.stdout.write( 

44 self.style.MIGRATE_HEADING( 

45 "\n4. Q-об'єкти — книги в наявності, окрім категорії 'Історія'" 

46 ) 

47 ) 

48 filtered = Book.objects.filter(Q(stock__gt=0) & ~Q(category__slug="history")) 

49 for book in filtered: 

50 self.stdout.write(f" - {book.title} ({book.category.name})") 

51 

52 self.stdout.write( 

53 self.style.MIGRATE_HEADING( 

54 "\n5. annotate() — кількість книг та середня ціна в кожній категорії" 

55 ) 

56 ) 

57 categories = Category.objects.annotate( 

58 book_count=Count("books"), avg_price=Avg("books__price") 

59 ) 

60 for category in categories: 

61 avg_price = f"{category.avg_price:.2f}" if category.avg_price else "—" 

62 self.stdout.write( 

63 f" - {category.name}: {category.book_count} книг, " 

64 f"середня ціна {avg_price} грн" 

65 ) 

66 

67 self.stdout.write( 

68 self.style.MIGRATE_HEADING( 

69 "\n6. annotate() + filter() — категорії, де є хоча б одна відсутня книга" 

70 ) 

71 ) 

72 categories_with_out_of_stock = Category.objects.annotate( 

73 out_of_stock_count=Count("books", filter=Q(books__stock=0)) 

74 ).filter(out_of_stock_count__gt=0) 

75 for category in categories_with_out_of_stock: 

76 self.stdout.write( 

77 f" - {category.name}: {category.out_of_stock_count} книг(и) відсутні" 

78 ) 

79 

80 self.stdout.write(self.style.SUCCESS("\nГотово."))
