1"""Кошик товарів, що зберігається в сесії користувача (request.session). 

2 

3У сесії лежить лише словник {id_книги: кількість} — без цін і назв, тому ціни 

4в кошику завжди актуальні (беруться з БД), а сама сесія залишається маленькою. 

5Ціна «фіксується» в момент оформлення замовлення (див. OrderItem.price). 

6"""

19class CartItem(TypedDict): 

20 """Один рядок кошика у вигляді, зручному для шаблону.""" 

21 

22 book: Book 

23 quantity: int 

24 price: Decimal 

25 total_price: Decimal 

26 over_stock: bool # у кошику більше, ніж зараз є на складі

29class Cart: 

30 def __init__(self, request: HttpRequest) -> None: 

31 self.session = request.session 

32 # .get(), а не setdefault(): так сесія не створюється для кожного 

33 # відвідувача, який ще нічого не додав у кошик. 

34 self.cart: dict[str, int] = dict(self.session.get(settings.CART_SESSION_ID) or {}) 

35 

36 def save(self) -> None: 

37 self.session[settings.CART_SESSION_ID] = self.cart 

38 self.session.modified = True 

39 

40 def add(self, book: Book, quantity: int = 1, override_quantity: bool = False) -> int: 

41 """Додає книгу в кошик і повертає підсумкову кількість цієї книги в кошику. 

42 

43 Args: 

44 book: книга. 

45 quantity: скільки додати (або яка кількість має бути, якщо override_quantity). 

46 override_quantity: True — замінити кількість, False — збільшити наявну. 

47 

48 Кількість обмежується залишком на складі. Якщо вийшло 0 (нема в наявності 

49 або кількість <= 0) — позиція видаляється з кошика. 

50 """ 

51 key = str(book.pk) 

52 current = self.cart.get(key, 0) 

53 wanted = quantity if override_quantity else current + quantity 

54 final = max(0, min(wanted, book.stock)) 

55 

56 if final == 0: 

57 self.cart.pop(key, None) 

58 else: 

59 self.cart[key] = final 

60 self.save() 

61 return final 

62 

63 def remove(self, book: Book) -> None: 

64 """Видаляє книгу з кошика повністю (незалежно від кількості).""" 

65 if self.cart.pop(str(book.pk), None) is not None: 

66 self.save() 

67 

68 def clear(self) -> None: 

69 """Повністю очищає кошик.""" 

70 self.cart = {} 

71 self.session.pop(settings.CART_SESSION_ID, None) 

72 self.session.modified = True 

73 

74 def lines(self) -> list[tuple[int, int]]: 

75 """Список (id_книги, кількість) — вхідні дані для створення замовлення.""" 

76 return [(int(book_id), qty) for book_id, qty in self.cart.items()]

78 def __iter__(self) -> Iterator[CartItem]: 

79 ids = [int(book_id) for book_id in self.cart] 

80 books = {b.pk: b for b in Book.objects.select_related("category").filter(pk__in=ids)} 

81 

82 # Книги, видалені з каталогу, тихо прибираємо з кошика. 

83 missing = [str(i) for i in ids if i not in books] 

84 if missing: 

85 for key in missing: 

86 self.cart.pop(key, None) 

87 self.save() 

88 

89 for book_id in ids: 

90 book = books.get(book_id) 

91 if book is None: 

92 continue 

93 quantity = self.cart[str(book_id)] 

94 yield CartItem( 

95 book=book, 

96 quantity=quantity, 

97 price=book.price, 

98 total_price=book.price * quantity, 

99 over_stock=quantity > book.stock, 

100 ) 

101 

102 def __len__(self) -> int: 

103 """Загальна кількість одиниць товару (без звернення до БД).""" 

104 return sum(self.cart.values())

106 def get_total_price(self) -> Decimal: 

107 return sum((item["total_price"] for item in self), Decimal("0"))
